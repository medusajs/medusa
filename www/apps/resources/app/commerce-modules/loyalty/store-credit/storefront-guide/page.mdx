# {metadata.title}

In this document, you'll learn how to integrate store credit functionality into your storefront using the Loyalty Plugin.

## Prerequisites

Before integrating store credit into your storefront, ensure you have:
- Installed and configured the Loyalty Plugin
- Access to Store API routes
- A Medusa storefront application with customer authentication
- Understanding of customer session management

## Store Credit Account Management

### Retrieving Customer Store Credit

Get the authenticated customer's store credit accounts:

```ts title="Get customer store credit"
// GET /store/store-credit-accounts
const getCustomerStoreCredit = async (customerId: string) => {
  try {
    const response = await fetch("/store/store-credit-accounts", {
      headers: {
        "Authorization": `Bearer ${customerToken}`,
        "Content-Type": "application/json",
      },
    })
    
    if (!response.ok) {
      throw new Error("Failed to fetch store credit accounts")
    }
    
    const { store_credit_accounts } = await response.json()
    
    return store_credit_accounts.map((account) => ({
      id: account.id,
      balance: account.balance,
      currency_code: account.currency_code,
      credits: account.credits,
      debits: account.debits,
    }))
  } catch (error) {
    console.error("Error fetching store credit:", error)
    return []
  }
}
```

## Cart Integration

### Applying Store Credit to Cart

Add store credit as payment to customer carts:

```ts title="Apply store credit to cart"
// POST /store/carts/{cart_id}/store-credits
const applyStoreCreditToCart = async (cartId: string, accountId: string, amount?: number) => {
  try {
    const body: any = { account_id: accountId }
    
    // If amount specified, use partial credit
    if (amount) {
      body.amount = amount
    }
    
    const response = await fetch(`/store/carts/${cartId}/store-credits`, {
      method: "POST",
      headers: {
        "Authorization": `Bearer ${customerToken}`,
        "Content-Type": "application/json",
      },
      body: JSON.stringify(body),
    })
    
    if (!response.ok) {
      const error = await response.json()
      throw new Error(error.message || "Failed to apply store credit")
    }
    
    const { cart } = await response.json()
    return cart
  } catch (error) {
    console.error("Error applying store credit:", error)
    throw error
  }
}
```

### Store Credit Validation

Validate store credit before applying:

```ts title="Store credit validation"
const validateStoreCredit = (account: any, cartTotal: number, requestedAmount?: number) => {
  // Check if account has balance
  if (account.balance <= 0) {
    return { valid: false, error: "No store credit balance available" }
  }
  
  // Check currency compatibility
  if (account.currency_code !== cart.currency_code) {
    return { 
      valid: false, 
      error: `Store credit currency (${account.currency_code}) does not match cart currency (${cart.currency_code})`, 
    }
  }
  
  // Validate requested amount
  if (requestedAmount) {
    if (requestedAmount > account.balance) {
      return { 
        valid: false, 
        error: `Requested amount exceeds available balance of $${account.balance / 100}`, 
      }
    }
    
    if (requestedAmount > cartTotal) {
      return { 
        valid: false, 
        error: "Requested amount exceeds cart total", 
      }
    }
  }
  
  return { valid: true }
}
```

## Store Credit UI Components

### Store Credit Balance Display

Show customer's available store credit:

```tsx title="components/StoreCreditBalance.tsx"
import { useState, useEffect } from "react"

interface StoreCreditBalanceProps {
  customerId: string
  onAccountSelect?: (account: any) => void
}

export const StoreCreditBalance: React.FC<StoreCreditBalanceProps> = ({ 
  customerId, 
  onAccountSelect, 
}) => {
  const [accounts, setAccounts] = useState([])
  const [isLoading, setIsLoading] = useState(true)

  useEffect(() => {
    const fetchAccounts = async () => {
      try {
        const storeCreditAccounts = await getCustomerStoreCredit(customerId)
        setAccounts(storeCreditAccounts)
      } catch (error) {
        console.error("Failed to load store credit:", error)
      } finally {
        setIsLoading(false)
      }
    }

    fetchAccounts()
  }, [customerId])

  const formatAmount = (amount: number, currency: string) => {
    return new Intl.NumberFormat("en-US", {
      style: "currency",
      currency: currency.toUpperCase(),
    }).format(amount / 100)
  }

  if (isLoading) {
    return <div>Loading store credit...</div>
  }

  if (accounts.length === 0) {
    return <div>No store credit available</div>
  }

  return (
    <div className="store-credit-balance">
      <h3>Available Store Credit</h3>
      {accounts.map((account) => (
        <div key={account.id} className="store-credit-account">
          <div className="account-info">
            <span className="balance">
              {formatAmount(account.balance, account.currency_code)}
            </span>
            <span className="currency">{account.currency_code.toUpperCase()}</span>
          </div>
          {onAccountSelect && account.balance > 0 && (
            <button
              onClick={() => onAccountSelect(account)}
              className="use-credit-button"
            >
              Use Credit
            </button>
          )}
        </div>
      ))}
    </div>
  )
}
```

### Store Credit Application Component

Allow customers to apply store credit with amount control:

```tsx title="components/StoreCreditApplication.tsx"
import { useState } from "react"

interface StoreCreditApplicationProps {
  account: any
  cartTotal: number
  onApply: (accountId: string, amount: number) => Promise<void>
  onCancel: () => void
}

export const StoreCreditApplication: React.FC<StoreCreditApplicationProps> = ({
  account,
  cartTotal,
  onApply,
  onCancel,
}) => {
  const [amount, setAmount] = useState(Math.min(account.balance, cartTotal))
  const [useFullBalance, setUseFullBalance] = useState(true)
  const [isLoading, setIsLoading] = useState(false)

  const maxAmount = Math.min(account.balance, cartTotal)

  const handleApply = async () => {
    setIsLoading(true)
    try {
      const applyAmount = useFullBalance ? maxAmount : amount
      await onApply(account.id, applyAmount)
    } catch (error) {
      console.error("Failed to apply store credit:", error)
    } finally {
      setIsLoading(false)
    }
  }

  const formatAmount = (value: number) => {
    return new Intl.NumberFormat("en-US", {
      style: "currency",
      currency: account.currency_code.toUpperCase(),
    }).format(value / 100)
  }

  return (
    <div className="store-credit-application">
      <h4>Apply Store Credit</h4>
      
      <div className="account-summary">
        <p>Available Balance: {formatAmount(account.balance)}</p>
        <p>Cart Total: {formatAmount(cartTotal)}</p>
        <p>Maximum Applicable: {formatAmount(maxAmount)}</p>
      </div>

      <div className="amount-selection">
        <label>
          <input
            type="radio"
            checked={useFullBalance}
            onChange={() => setUseFullBalance(true)}
          />
          Use maximum amount ({formatAmount(maxAmount)})
        </label>
        
        <label>
          <input
            type="radio"
            checked={!useFullBalance}
            onChange={() => setUseFullBalance(false)}
          />
          Use custom amount
        </label>
        
        {!useFullBalance && (
          <div className="custom-amount">
            <input
              type="number"
              min={1}
              max={maxAmount}
              value={amount}
              onChange={(e) => setAmount(parseInt(e.target.value) || 0)}
              step={1}
            />
            <span>cents</span>
          </div>
        )}
      </div>

      <div className="actions">
        <button
          onClick={handleApply}
          disabled={isLoading || (!useFullBalance && (amount < 1 || amount > maxAmount))}
        >
          {isLoading ? "Applying..." : `Apply ${formatAmount(useFullBalance ? maxAmount : amount)}`}
        </button>
        <button onClick={onCancel} disabled={isLoading}>
          Cancel
        </button>
      </div>
    </div>
  )
}
```

## Account Activity and History

### Transaction History Component

Display store credit transaction history:

```tsx title="components/StoreCreditHistory.tsx"
import { useState, useEffect } from "react"

interface StoreCreditHistoryProps {
  accountId: string
}

export const StoreCreditHistory: React.FC<StoreCreditHistoryProps> = ({ accountId }) => {
  const [transactions, setTransactions] = useState([])
  const [isLoading, setIsLoading] = useState(true)

  useEffect(() => {
    const fetchTransactions = async () => {
      try {
        const response = await fetch(`/store/store-credit-accounts/${accountId}/transactions`, {
          headers: {
            "Authorization": `Bearer ${customerToken}`,
          },
        })
        
        const { transactions } = await response.json()
        setTransactions(transactions)
      } catch (error) {
        console.error("Failed to load transaction history:", error)
      } finally {
        setIsLoading(false)
      }
    }

    fetchTransactions()
  }, [accountId])

  const formatAmount = (amount: number, type: string, currency: string) => {
    const formatted = new Intl.NumberFormat("en-US", {
      style: "currency",
      currency: currency.toUpperCase(),
    }).format(amount / 100)
    
    return type === "credit" ? `+${formatted}` : `-${formatted}`
  }

  const formatDate = (dateString: string) => {
    return new Intl.DateTimeFormat("en-US", {
      year: "numeric",
      month: "short",
      day: "numeric",
      hour: "2-digit",
      minute: "2-digit",
    }).format(new Date(dateString))
  }

  if (isLoading) {
    return <div>Loading transaction history...</div>
  }

  return (
    <div className="store-credit-history">
      <h3>Transaction History</h3>
      {transactions.length === 0 ? (
        <p>No transactions found</p>
      ) : (
        <div className="transactions">
          {transactions.map((transaction) => (
            <div key={transaction.id} className={`transaction ${transaction.type}`}>
              <div className="transaction-details">
                <div className="amount">
                  {formatAmount(transaction.amount, transaction.type, transaction.currency_code)}
                </div>
                <div className="info">
                  <div className="reference">{transaction.reference}</div>
                  {transaction.note && (
                    <div className="note">{transaction.note}</div>
                  )}
                  <div className="date">{formatDate(transaction.created_at)}</div>
                </div>
              </div>
            </div>
          ))}
        </div>
      )}
    </div>
  )
}
```

## Claiming Store Credit

### Store Credit Claim Component

Allow customers to claim store credit using codes or links:

```tsx title="components/StoreCreditClaim.tsx"
import { useState } from "react"

export const StoreCreditClaim: React.FC = () => {
  const [claimCode, setClaimCode] = useState("")
  const [isLoading, setIsLoading] = useState(false)
  const [result, setResult] = useState<{ success: boolean; message: string } | null>(null)

  const handleClaim = async (e: React.FormEvent) => {
    e.preventDefault()
    
    if (!claimCode.trim()) {
      setResult({ success: false, message: "Please enter a claim code" })
      return
    }

    setIsLoading(true)
    setResult(null)

    try {
      const response = await fetch("/store/store-credit-accounts/claim", {
        method: "POST",
        headers: {
          "Authorization": `Bearer ${customerToken}`,
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          code: claimCode.trim(),
        }),
      })

      if (!response.ok) {
        const error = await response.json()
        throw new Error(error.message || "Failed to claim store credit")
      }

      const { store_credit_account } = await response.json()
      setResult({
        success: true,
        message: `Successfully claimed $${store_credit_account.balance / 100} in store credit!`,
      })
      setClaimCode("")
    } catch (error) {
      setResult({
        success: false,
        message: error.message,
      })
    } finally {
      setIsLoading(false)
    }
  }

  return (
    <div className="store-credit-claim">
      <h3>Claim Store Credit</h3>
      
      <form onSubmit={handleClaim}>
        <div className="form-group">
          <label htmlFor="claim-code">Claim Code</label>
          <input
            id="claim-code"
            type="text"
            value={claimCode}
            onChange={(e) => setClaimCode(e.target.value)}
            placeholder="Enter your claim code"
            disabled={isLoading}
          />
        </div>
        
        <button type="submit" disabled={isLoading || !claimCode.trim()}>
          {isLoading ? "Claiming..." : "Claim Store Credit"}
        </button>
      </form>

      {result && (
        <div className={`result ${result.success ? "success" : "error"}`}>
          {result.message}
        </div>
      )}
    </div>
  )
}
```

## Checkout Integration

### Complete Checkout with Store Credit

Integrate store credit into the checkout flow:

```tsx title="components/CheckoutStoreCreditSection.tsx"
import { useState } from "react"
import { StoreCreditBalance } from "./StoreCreditBalance"
import { StoreCreditApplication } from "./StoreCreditApplication"
import { AppliedStoreCredits } from "./AppliedStoreCredits"

interface CheckoutStoreCreditSectionProps {
  cart: any
  customerId: string
  onCartUpdate: (cart: any) => void
}

export const CheckoutStoreCreditSection: React.FC<CheckoutStoreCreditSectionProps> = ({
  cart,
  customerId,
  onCartUpdate,
}) => {
  const [selectedAccount, setSelectedAccount] = useState(null)
  const [appliedCredits, setAppliedCredits] = useState(cart.store_credit_transactions || [])

  const handleApplyStoreCredit = async (accountId: string, amount: number) => {
    try {
      const updatedCart = await applyStoreCreditToCart(cart.id, accountId, amount)
      onCartUpdate(updatedCart)
      setSelectedAccount(null)
    } catch (error) {
      console.error("Failed to apply store credit:", error)
    }
  }

  const handleRemoveStoreCredit = async (accountId: string) => {
    try {
      const response = await fetch(`/store/carts/${cart.id}/store-credits/${accountId}`, {
        method: "DELETE",
        headers: {
          "Authorization": `Bearer ${customerToken}`,
        },
      })
      
      const { cart: updatedCart } = await response.json()
      onCartUpdate(updatedCart)
    } catch (error) {
      console.error("Failed to remove store credit:", error)
    }
  }

  const totalStoreCreditApplied = appliedCredits.reduce(
    (sum, credit) => sum + credit.amount_applied, 
    0
  )

  const remainingTotal = cart.total - totalStoreCreditApplied

  return (
    <div className="checkout-store-credit">
      <h3>Store Credit</h3>
      
      <AppliedStoreCredits
        storeCredits={appliedCredits}
        onRemove={handleRemoveStoreCredit}
      />

      {!selectedAccount && remainingTotal > 0 && (
        <StoreCreditBalance
          customerId={customerId}
          onAccountSelect={setSelectedAccount}
        />
      )}

      {selectedAccount && (
        <StoreCreditApplication
          account={selectedAccount}
          cartTotal={remainingTotal}
          onApply={handleApplyStoreCredit}
          onCancel={() => setSelectedAccount(null)}
        />
      )}

      {remainingTotal <= 0 && (
        <div className="fully-covered">
          <p>Order fully covered by store credit!</p>
        </div>
      )}
    </div>
  )
}
```

## Best Practices

### User Experience
- Clearly display available store credit balances
- Allow partial application of store credit
- Show remaining balance after application
- Provide transaction history for transparency

### Security
- Always authenticate users before showing store credit
- Validate all operations server-side
- Use secure token-based authentication
- Log all store credit operations

### Error Handling
- Handle network errors gracefully
- Provide clear error messages
- Implement retry mechanisms for failed operations
- Validate user inputs client-side

### Performance
- Cache store credit balances appropriately
- Implement loading states for async operations
- Use pagination for transaction history
- Optimize API calls to reduce latency
